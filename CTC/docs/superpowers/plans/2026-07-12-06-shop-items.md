```csharp
        {
```

```csharp
        [Rpc(SendTo.Server)]
        private void FireRpc(Vector3 _direction)
        {
            WeaponStats stats = ShopLogic.WeaponStatsForTier(m_Tier.Value);

            double now = NetworkManager.ServerTime.Time;
            if (!CombatLogic.IsCooldownReady(m_LastUseTime, now, stats.CooldownSeconds)) return;
            m_LastUseTime = now;

            // 원점을 전진시키면 밀착한 고양이 콜라이더 '내부'에서 시작해 빗나간다(리뷰 I-1).
            // 자기 위치에서 RaycastAll 후 자기 자신만 스킵하고 최근접 유효 히트를 취한다.
            Vector3 origin = transform.position + Vector3.up * MUZZLE_HEIGHT;
            Vector3 end = origin + _direction * stats.Range;
            RaycastHit[] hits = Physics.RaycastAll(origin, _direction, stats.Range);
            System.Array.Sort(hits, (_a, _b) => _a.distance.CompareTo(_b.distance));

            foreach (RaycastHit hit in hits)
            {
                if (hit.collider.transform.root == transform.root) continue; // 자기 몸 스킵

                Health target = hit.collider.GetComponentInParent<Health>();
                if (target == null)
                {
                    end = hit.point; // 벽/가구가 먼저 막음
                    break;
                }
                if (target.IsDead) continue;

                end = hit.point;

                RoleTag tag = target.GetComponent<RoleTag>();
                if (tag == null || tag.Role != RoleId.Cat) break; // 쥐 등 비대상이 막음

                target.TakeDamage(stats.Damage);
                break;
            }

            TracerRpc(origin, end);
        }
```

```csharp
namespace CTC.Runtime.Player

    /// <summary>
    /// 대시(상점 구매 영구 능력). 소유자가 Shift를 누르면 몸 전방으로 짧게 돌진.
    /// 이동은 소유자 권위(ClientNetworkTransform)라 CC.Move로 로컬 실행만 하면 동기화된다.
    /// </summary>

    public class MouseDash : NetworkBehaviour
    {
        private const float DASH_DISTANCE = 4f;
        private const float DASH_DURATION_SECONDS = 0.2f;
        private const float COOLDOWN_SECONDS = 5f;

        // 구매 여부. 서버 기록(ShopController), 소유자 읽기.
        private readonly NetworkVariable<bool> m_Owned = new NetworkVariable<bool>();

        private CharacterController m_Controller;
        private Vector3 m_DashDirection;
        private float m_DashRemainingSeconds;
        private double m_LastUseTime = -999.0;

        public bool Owned => m_Owned.Value;

        /// <summary>서버 전용(ShopController).</summary>
        public void ServerSetOwned()
        {
            if (IsServer) m_Owned.Value = true;
        }

        private void Awake()
        {
            m_Controller = GetComponent<CharacterController>();
        }

        private void Update()
        {
            if (!IsOwner) return;

            if (m_DashRemainingSeconds > 0f)
            {
                float dt = Mathf.Min(Time.deltaTime, m_DashRemainingSeconds);
                m_Controller.Move(m_DashDirection * (DASH_DISTANCE / DASH_DURATION_SECONDS) * dt);
                m_DashRemainingSeconds -= Time.deltaTime;
                return;
            }

            if (!m_Owned.Value) return;

            Keyboard kb = Keyboard.current;
            if (kb == null || !kb.leftShiftKey.wasPressedThisFrame) return;
            if (!Combat.CombatLogic.IsCooldownReady(m_LastUseTime, Time.timeAsDouble, COOLDOWN_SECONDS)) return;

            m_LastUseTime = Time.timeAsDouble;
            m_DashDirection = new Vector3(transform.forward.x, 0f, transform.forward.z).normalized;
            m_DashRemainingSeconds = DASH_DURATION_SECONDS;
        }
```

```csharp
using CTC.Runtime.Combat;
using CTC.Runtime.Lobby;
using Unity.Netcode;
using UnityEngine;

namespace CTC.Runtime.Economy
{
    /// <summary>
    /// 구매 진입점(플레이어 프리팹). 소유자 Buy → ServerRpc → 서버가
    /// 역할·적용가능 검증 → TrySpend → 효과 적용/설치물 스폰. 검증 실패 시 차감 없음.
    /// </summary>
    public class ShopController : NetworkBehaviour
    {
        private const float PLACE_FORWARD_METERS = 1.5f;

        [SerializeField] private GameObject m_DecoyPrefab;
        [SerializeField] private GameObject m_FortressPrefab;

        /// <summary>로컬 UI/QA 진입점.</summary>
        public void Buy(ShopItemId _itemId)
        {
            BuyRpc((int)_itemId);
        }

        /// <summary>구매 가능 여부(로컬 UI 표시용 — 판정 권위는 서버 BuyRpc).</summary>
        public bool CanApply(ShopItemId _itemId)
        {
            switch (_itemId)
            {
                case ShopItemId.Slingshot:
                    MousePopgun gun1 = GetComponent<MousePopgun>();
                    return gun1 != null && gun1.Tier < 1;
                case ShopItemId.MachineGun:
                    MousePopgun gun2 = GetComponent<MousePopgun>();
                    return gun2 != null && gun2.Tier < 2;
                case ShopItemId.Dash:
                    MouseDash dash = GetComponent<MouseDash>();
                    return dash != null && !dash.Owned;
                case ShopItemId.Decoy:
                    return m_DecoyPrefab != null;
                case ShopItemId.Fortress:
                    return m_FortressPrefab != null;
                case ShopItemId.CatAttack:
                    CatPounce pounce = GetComponent<CatPounce>();
                    return pounce != null && ShopLogic.CanAddStack(pounce.DamageStacks);
                case ShopItemId.CatSpeed:
                    Player.PlayerMovement move = GetComponent<Player.PlayerMovement>();
                    return move != null && ShopLogic.CanAddStack(move.SpeedStacks);
                default:
                    return false;
            }
        }

        [Rpc(SendTo.Server)]
        private void BuyRpc(int _itemId)
        {
            ShopItem item = ShopLogic.Find((ShopItemId)_itemId);
            if (item.Id == ShopItemId.None) return;

            RoleTag tag = GetComponent<RoleTag>();
            if (tag == null || tag.Role != item.Role) return;

            if (!CanApply(item.Id)) return;

            CheeseWallet wallet = GetComponent<CheeseWallet>();
            if (wallet == null || !wallet.TrySpend(item.Cost)) return;

            Apply(item.Id);
            Debug.Log($"[ShopController] {OwnerClientId} 구매: {item.Id} (-{item.Cost})");
        }

        // 서버 전용 — BuyRpc에서 검증·차감 후 호출.
        private void Apply(ShopItemId _itemId)
        {
            switch (_itemId)
            {
                case ShopItemId.Slingshot:
                    GetComponent<MousePopgun>().ServerSetTier(1);
                    break;
                case ShopItemId.MachineGun:
                    GetComponent<MousePopgun>().ServerSetTier(2);
                    break;
                case ShopItemId.Dash:
                    GetComponent<MouseDash>().ServerSetOwned();
                    break;
                case ShopItemId.Decoy:
                    SpawnStructure(m_DecoyPrefab);
                    break;
                case ShopItemId.Fortress:
                    SpawnStructure(m_FortressPrefab);
                    break;
                case ShopItemId.CatAttack:
                    GetComponent<CatPounce>().ServerAddDamageStack();
                    break;
                case ShopItemId.CatSpeed:
                    GetComponent<Player.PlayerMovement>().ServerAddSpeedStack();
                    break;
            }
        }

        private void SpawnStructure(GameObject _prefab)
        {
            Vector3 forward = new Vector3(transform.forward.x, 0f, transform.forward.z).normalized;
            Vector3 pos = transform.position + forward * PLACE_FORWARD_METERS;
            GameObject go = Instantiate(_prefab, pos, Quaternion.identity);
            go.GetComponent<NetworkObject>().Spawn(true);
        }
    }
}
```

```csharp
                body.transform.localPosition = new Vector3(0f, scale.y, 0f);

                BakeGlobalObjectIdHash(asset.GetComponent<NetworkObject>());
```
